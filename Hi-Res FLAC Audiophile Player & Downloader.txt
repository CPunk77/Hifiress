export default function App() {
  // Навигация по вкладкам
  const [activeTab, setActiveTab] = useState('player'); // 'player', 'recommendations', 'search', 'library', 'playlists', 'customize', 'code'
  
  // Состояние аудиоплеера
  const [library, setLibrary] = useState(INITIAL_LIBRARY);
  const [currentTrackIndex, setCurrentTrackIndex] = useState(0);
  const [isPlaying, setIsPlaying] = useState(false);
  const [currentTime, setCurrentTime] = useState(42);
  const [isShuffle, setIsShuffle] = useState(false);
  const [isRepeat, setIsRepeat] = useState(false);
  const [volume, setVolume] = useState(85);
  
  // Плейлисты
  const [playlists, setPlaylists] = useState([
    { id: 'pl-1', name: 'Избранные FLAC', tracks: ['flac-1', 'flac-2'], icon: 'Heart' },
    { id: 'pl-2', name: 'Аудиофильский Ночь', tracks: ['flac-1', 'flac-2', 'flac-3'], icon: 'Zap' },
    { id: 'pl-3', name: 'Рок Легенды', tracks: ['flac-1', 'flac-2'], icon: 'Music' }
  ]);
  const [newPlaylistName, setNewPlaylistName] = useState('');
  const [showCreatePlaylist, setShowCreatePlaylist] = useState(false);

  // Эквалайзер
  const [eqPreset, setEqPreset] = useState('Аудиофил (Flat)');
  const [eqBands, setEqBands] = useState([0, 0, 0, 0, 0]);
  const [showEqModal, setShowEqModal] = useState(false);

  // Поиск и Загрузки
  const [searchQuery, setSearchQuery] = useState('');
  const [searchResults, setSearchResults] = useState(SEARCH_RESULTS_DATABASE);
  const [downloads, setDownloads] = useState([]);

  // Предпочтения пользователя (Интересы)
  const [userInterests, setUserInterests] = useState(['Rock', 'Electronic', 'Hi-Res 24-bit', 'Synthwave']);

  // Кастомизация темы
  const [themePreset, setThemePreset] = useState('amoled');
  const [accentColor, setAccentColor] = useState('#6366f1');
  const [coverEffect, setCoverEffect] = useState('glow');
  const [fontFamily, setFontFamily] = useState('sans');

  const currentTrack = library[currentTrackIndex] || INITIAL_LIBRARY[0];

  // Симуляция таймера воспроизведения
  useEffect(() => {
    let interval = null;
    if (isPlaying) {
      interval = setInterval(() => {
        setCurrentTime((prev) => {
          if (prev >= currentTrack.durationSec) {
            handleNextTrack();
            return 0;
          }
          return prev + 1;
        });
      }, 1000);
    } else {
      clearInterval(interval);
    }
    return () => clearInterval(interval);
  }, [isPlaying, currentTrack]);

  // Симуляция фоновых скачиваний
  useEffect(() => {
    const activeInterval = setInterval(() => {
      setDownloads((prevDownloads) => {
        return prevDownloads.map((item) => {
          if (item.progress < 100) {
            const nextProgress = Math.min(100, item.progress + Math.floor(Math.random() * 15) + 8);
            if (nextProgress === 100) {
              setLibrary((prevLib) => {
                if (prevLib.some((t) => t.id === item.id)) return prevLib;
                return [...prevLib, { ...item, downloaded: true, progress: 100, favorite: false, isHiRes: true }];
              });
            }
            return {
              ...item,
              progress: nextProgress,
              speed: `${(Math.random() * 5 + 10).toFixed(1)} МБ/с`
            };
          }
          return item;
        });
      });
    }, 700);

    return () => clearInterval(activeInterval);
  }, []);

  const handleNextTrack = () => {
    if (isShuffle) {
      const nextIndex = Math.floor(Math.random() * library.length);
      setCurrentTrackIndex(nextIndex);
    } else {
      setCurrentTrackIndex((prev) => (prev + 1) % library.length);
    }
    setCurrentTime(0);
  };

  const handlePrevTrack = () => {
    setCurrentTrackIndex((prev) => (prev === 0 ? library.length - 1 : prev - 1));
    setCurrentTime(0);
  };

  const toggleFavorite = (trackId) => {
    setLibrary(prev => prev.map(t => t.id === trackId ? { ...t, favorite: !t.favorite } : t));
  };

  const startDownload = (track) => {
    if (downloads.some((d) => d.id === track.id) || library.some((l) => l.id === track.id)) {
      return;
    }
    const newDownloadItem = {
      ...track,
      progress: 0,
      speed: '14.2 МБ/с'
    };
    setDownloads((prev) => [newDownloadItem, ...prev]);
  };

  const createPlaylist = () => {
    if (!newPlaylistName.trim()) return;
    const newPl = {
      id: `pl-${Date.now()}`,
      name: newPlaylistName,
      tracks: [],
      icon: 'Music'
    };
    setPlaylists([...playlists, newPl]);
    setNewPlaylistName('');
    setShowCreatePlaylist(false);
  };

  const applyEqPreset = (name) => {
    setEqPreset(name);
    switch (name) {
      case 'Усиление баса': setEqBands([6, 4, 1, -1, 2]); break;
      case 'Чистый вокал': setEqBands([-2, 1, 5, 4, 2]); break;
      case 'Акустика': setEqBands([3, 2, 0, 2, 4]); break;
      case 'Электроника': setEqBands([5, 3, -1, 2, 5]); break;
      default: setEqBands([0, 0, 0, 0, 0]); break;
    }
  };

  const getThemeBg = () => {
    switch (themePreset) {
      case 'amoled': return 'bg-black text-white';
      case 'cyberpunk': return 'bg-slate-950 text-cyan-400 border-cyan-500/30';
      case 'retro': return 'bg-amber-950/90 text-amber-200';
      case 'minimal': return 'bg-neutral-900 text-neutral-100';
      default: return 'bg-black text-white';
    }
  };

  const formatTime = (seconds) => {
    const mins = Math.floor(seconds / 60);
    const secs = Math.floor(seconds % 60);
    return `${mins}:${secs < 10 ? '0' : ''}${secs}`;
  };

  return (
    <div className={`w-full min-h-screen flex flex-col justify-center items-center p-2 sm:p-6 transition-colors duration-300 font-${fontFamily} ${getThemeBg()}`}>
      {/* Корпус Мобильного Устройства */}
      <div className="w-full max-w-md h-[900px] bg-neutral-900 border border-neutral-800 rounded-[42px] shadow-2xl flex flex-col overflow-hidden relative border-opacity-60 ring-1 ring-white/10">
        
        {/* Верхняя строка статуса Android */}
        <div className="w-full h-8 px-6 pt-2 flex justify-between items-center text-xs text-neutral-400 font-mono select-none z-30">
          <span>12:45</span>
          <div className="flex items-center space-x-2">
            <span className="text-[10px] bg-amber-500/20 text-amber-400 border border-amber-500/30 px-1 rounded font-bold">FLAC Hi-Res</span>
            <WifiIcon className="w-3.5 h-3.5" />
            <BatteryIcon className="w-4 h-4" />
          </div>
        </div>

        {/* Шапка приложения */}
        <div className="px-5 py-3 flex justify-between items-center z-20 border-b border-white/5 bg-black/30 backdrop-blur-md">
          <div className="flex items-center space-x-2">
            <div className="p-1.5 rounded-lg" style={{ backgroundColor: `${accentColor}25` }}>
              <Radio className="w-5 h-5" style={{ color: accentColor }} />
            </div>
            <div>
              <h1 className="text-sm font-bold tracking-wide flex items-center gap-1.5">
                SONIC FLAC
                <span className="text-[9px] px-1.5 py-0.5 rounded-full bg-emerald-500/20 text-emerald-400 border border-emerald-500/30 font-mono">24-BIT</span>
              </h1>
              <p className="text-[10px] text-neutral-400">Прямой lossless аудио-поток</p>
            </div>
          </div>
          
          <button 
            onClick={() => setShowEqModal(true)}
            className="p-2 rounded-full hover:bg-white/10 transition-colors border border-white/10 flex items-center justify-center text-neutral-300"
            title="Эквалайзер"
          >
            <Sliders className="w-4 h-4" />
          </button>
        </div>

        {/* Главная Рабочая Область */}
        <div className="flex-1 overflow-y-auto relative scrollbar-none">
          {activeTab === 'player' && (
            <PlayerTab 
              currentTrack={currentTrack}
              isPlaying={isPlaying}
              setIsPlaying={setIsPlaying}
              currentTime={currentTime}
              setCurrentTime={setCurrentTime}
              handleNextTrack={handleNextTrack}
              handlePrevTrack={handlePrevTrack}
              isShuffle={isShuffle}
              setIsShuffle={setIsShuffle}
              isRepeat={isRepeat}
              setIsRepeat={setIsRepeat}
              accentColor={accentColor}
              coverEffect={coverEffect}
              volume={volume}
              setVolume={setVolume}
              toggleFavorite={toggleFavorite}
              formatTime={formatTime}
            />
          )}

          {activeTab === 'recommendations' && (
            <RecommendationsTab 
              userInterests={userInterests}
              setUserInterests={setUserInterests}
              recommendations={RECOMMENDATIONS_CATALOG}
              startDownload={startDownload}
              accentColor={accentColor}
              downloads={downloads}
              library={library}
            />
          )}

          {activeTab === 'search' && (
            <SearchTab 
              searchQuery={searchQuery}
              setSearchQuery={setSearchQuery}
              searchResults={searchResults}
              downloads={downloads}
              startDownload={startDownload}
              accentColor={accentColor}
              library={library}
            />
          )}

          {activeTab === 'library' && (
            <LibraryTab 
              library={library}
              currentTrack={currentTrack}
              setCurrentTrackIndex={setCurrentTrackIndex}
              setIsPlaying={setIsPlaying}
              accentColor={accentColor}
              toggleFavorite={toggleFavorite}
            />
          )}

          {activeTab === 'playlists' && (
            <PlaylistsTab 
              playlists={playlists}
              library={library}
              setCurrentTrackIndex={setCurrentTrackIndex}
              setIsPlaying={setIsPlaying}
              showCreatePlaylist={showCreatePlaylist}
              setShowCreatePlaylist={setShowCreatePlaylist}
              newPlaylistName={newPlaylistName}
              setNewPlaylistName={setNewPlaylistName}
              createPlaylist={createPlaylist}
              accentColor={accentColor}
            />
          )}

          {activeTab === 'customize' && (
            <CustomizeTab 
              themePreset={themePreset}
              setThemePreset={setThemePreset}
              accentColor={accentColor}
              setAccentColor={setAccentColor}
              coverEffect={coverEffect}
              setCoverEffect={setCoverEffect}
              fontFamily={fontFamily}
              setFontFamily={setFontFamily}
            />
          )}

          {activeTab === 'code' && (
            <CodeViewerTab accentColor={accentColor} />
          )}
        </div>

        {/* Модальное окно Эквалайзера */}
        {showEqModal && (
          <EqualizerModal 
            onClose={() => setShowEqModal(false)}
            eqPreset={eqPreset}
            applyEqPreset={applyEqPreset}
            eqBands={eqBands}
            setEqBands={setEqBands}
            accentColor={accentColor}
          />
        )}

        {/* Нижняя Навигационная Панель */}
        <div className="w-full bg-neutral-950/95 border-t border-white/10 px-1 py-2 flex justify-around items-center z-30 backdrop-blur-xl">
          <NavButton 
            active={activeTab === 'player'} 
            onClick={() => setActiveTab('player')} 
            icon={<Disc className="w-4 h-4" />} 
            label="Плеер"
            accentColor={accentColor}
          />
          <NavButton 
            active={activeTab === 'recommendations'} 
            onClick={() => setActiveTab('recommendations')} 
            icon={<Compass className="w-4 h-4" />} 
            label="Микс"
            accentColor={accentColor}
          />
          <NavButton 
            active={activeTab === 'search'} 
            onClick={() => setActiveTab('search')} 
            icon={<Search className="w-4 h-4" />} 
            label="Поиск"
            accentColor={accentColor}
            badge={downloads.filter(d => d.progress < 100).length}
          />
          <NavButton 
            active={activeTab === 'library'} 
            onClick={() => setActiveTab('library')} 
            icon={<HardDrive className="w-4 h-4" />} 
            label="Треки"
            accentColor={accentColor}
            badge={library.length}
          />
          <NavButton 
            active={activeTab === 'playlists'} 
            onClick={() => setActiveTab('playlists')} 
            icon={<ListMusic className="w-4 h-4" />} 
            label="Списки"
            accentColor={accentColor}
          />
          <NavButton 
            active={activeTab === 'customize'} 
            onClick={() => setActiveTab('customize')} 
            icon={<Palette className="w-4 h-4" />} 
            label="Стиль"
            accentColor={accentColor}
          />
        </div>

      </div>
    </div>
  );
}

// ==========================================
// ВКЛАДКИ ПРИЛОЖЕНИЯ
// ==========================================

function PlayerTab({ 
  currentTrack, isPlaying, setIsPlaying, currentTime, setCurrentTime, 
  handleNextTrack, handlePrevTrack, isShuffle, setIsShuffle, 
  isRepeat, setIsRepeat, accentColor, coverEffect, volume, setVolume, toggleFavorite, formatTime 
}) {
  return (
    <div className="p-6 flex flex-col items-center justify-between min-h-full space-y-4">
      
      {/* Характеристики аудиопотока */}
      <div className="w-full flex justify-between items-center text-xs bg-white/5 border border-white/10 p-2.5 rounded-xl backdrop-blur-md">
        <div className="flex items-center space-x-2">
          <Zap className="w-4 h-4 text-amber-400 animate-pulse" />
          <div>
            <span className="font-bold text-neutral-200">{currentTrack.format}</span>
            <span className="text-neutral-400 text-[11px] block">{currentTrack.sampleRate}</span>
          </div>
        </div>
        <div className="text-right">
          <span className="font-mono text-xs px-2 py-0.5 rounded bg-black/40 border border-white/10 text-emerald-400">
            {currentTrack.bitrate}
          </span>
        </div>
      </div>

      {/* Обложка альбома с кастомным оформлением */}
      <div className="relative w-60 h-60 my-1 flex items-center justify-center">
        {coverEffect === 'glow' && (
          <div 
            className="absolute inset-0 rounded-3xl blur-2xl opacity-50 transition-all duration-700"
            style={{ 
              backgroundImage: `url(${currentTrack.cover})`,
              backgroundSize: 'cover'
            }}
          />
        )}

        {coverEffect === 'vinyl' && (
          <div className={`absolute w-56 h-56 rounded-full border-4 border-neutral-800 bg-neutral-950 shadow-2xl flex items-center justify-center ${isPlaying ? 'animate-spin' : ''}`} style={{ animationDuration: '8s' }}>
            <div className="w-20 h-20 rounded-full border border-neutral-700 bg-neutral-900 flex items-center justify-center">
              <div className="w-6 h-6 rounded-full bg-neutral-950 border border-neutral-800" />
            </div>
          </div>
        )}

        <div className={`relative z-10 w-52 h-52 rounded-2xl overflow-hidden shadow-2xl border border-white/10 transition-transform duration-300 ${isPlaying ? 'scale-105' : 'scale-95'} ${coverEffect === 'glass' ? 'backdrop-blur-md bg-white/10 p-2' : ''}`}>
          <img 
            src={currentTrack.cover} 
            alt={currentTrack.title} 
            className="w-full h-full object-cover rounded-xl"
          />
        </div>
      </div>

      {/* Анимированный Спектроанализатор */}
      <div className="w-full h-8 flex items-end justify-center space-x-1.5 px-4">
        {[40, 75, 30, 90, 60, 100, 45, 80, 55, 95, 35, 70, 85, 50, 65, 90, 40, 80].map((height, i) => (
          <div 
            key={i} 
            className="w-1.5 rounded-t-full transition-all duration-200"
            style={{ 
              height: isPlaying ? `${Math.max(15, (height * (Math.sin(currentTime + i) + 1.5)) / 2.5)}%` : '15%',
              backgroundColor: accentColor,
              opacity: isPlaying ? 0.9 : 0.3
            }}
          />
        ))}
      </div>

      {/* Метаданные трека */}
      <div className="w-full flex justify-between items-center px-2">
        <div className="text-left space-y-0.5 overflow-hidden pr-2">
          <h2 className="text-base font-bold truncate text-white tracking-wide">{currentTrack.title}</h2>
          <p className="text-xs text-neutral-400 truncate">{currentTrack.artist} — <span className="text-neutral-500">{currentTrack.album}</span></p>
        </div>
        <button 
          onClick={() => toggleFavorite(currentTrack.id)}
          className="p-2 text-neutral-400 hover:text-red-500 transition-colors"
        >
          <Heart className={`w-5 h-5 ${currentTrack.favorite ? 'fill-red-500 text-red-500' : ''}`} />
        </button>
      </div>

      {/* Полоса прокрутки */}
      <div className="w-full space-y-1 px-2">
        <input 
          type="range" 
          min="0" 
          max={currentTrack.durationSec} 
          value={currentTime} 
          onChange={(e) => setCurrentTime(Number(e.target.value))}
          className="w-full h-1.5 bg-neutral-800 rounded-lg appearance-none cursor-pointer accent-indigo-500"
          style={{ accentColor: accentColor }}
        />
        <div className="flex justify-between text-[11px] text-neutral-400 font-mono">
          <span>{formatTime(currentTime)}</span>
          <span>{currentTrack.duration}</span>

function CodeViewerTab() {
  const [copied, setCopied] = useState(false);

  const kotlinSnippet = `// Android Jetpack Compose + ExoPlayer FLAC Модуль
package com.sonic.flacplayer

import android.net.Uri
import androidx.media3.common.MediaItem
import androidx.media3.common.MimeTypes
import androidx.media3.exoplayer.ExoPlayer

class AudioPlayerEngine(private val context: Context) {
    private var player: ExoPlayer = ExoPlayer.Builder(context).build()

    fun playFlacStream(url: String) {
        val mediaItem = MediaItem.Builder()
            .setUri(Uri.parse(url))
            .setMimeType(MimeTypes.AUDIO_FLAC)
            .build()
        
        player.setMediaItem(mediaItem)
        player.prepare()
        player.play()
    }
}`;

  const copyCode = () => {
    navigator.clipboard.writeText(kotlinSnippet);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className="p-4 space-y-4">
      <div className="flex justify-between items-center">
        <div>
          <h2 className="text-base font-bold text-white">Android Нативный Код</h2>

        </div>
        <button 
          onClick={copyCode}
          className="px-3 py-1.5 bg-white/10 hover:bg-white/20 border border-white/10 rounded-lg text-xs font-mono text-white flex items-center gap-1.5 transition-colors"
        >
          {copied ? <Check className="w-3.5 h-3.5 text-emerald-400" /> : <Copy className="w-3.5 h-3.5" />}
          <span>{copied ? 'Скопировано' : 'Скопировать'}</span>
        </button>
      </div>

      <div className="bg-black/80 border border-white/10 rounded-xl p-3 overflow-x-auto font-mono text-[11px] text-emerald-400 leading-relaxed shadow-inner">

      </div>

  );
}

    <div className="absolute inset-0 bg-black/80 backdrop-blur-md z-50 flex items-center justify-center p-4">
      <div className="w-full bg-neutral-900 border border-white/10 rounded-2xl p-5 space-y-4 shadow-2xl">
        <div className="flex justify-between items-center border-b border-white/10 pb-3">
          <h3 className="text-sm font-bold text-white flex items-center gap-2">
            <Sliders className="w-4 h-4 text-indigo-400" />
            5-Полосный Эквалайзер
          </h3>
          <button onClick={onClose} className="text-neutral-400 hover:text-white text-xs font-bold px-2 py-1 bg-white/5 rounded-lg">Закрыть</button>
        </div>

        <div className="space-y-1.5">
          <label className="text-[11px] font-bold text-neutral-400 uppercase">Пресеты звука</label>
          <div className="flex flex-wrap gap-1.5">
            {['Аудиофил (Flat)', 'Усиление баса', 'Чистый вокал', 'Акустика', 'Электроника'].map((p) => (
              <button
                key={p}
                onClick={() => applyEqPreset(p)}
                className={`px-2.5 py-1 rounded-lg text-xs font-medium transition-colors ${
                  eqPreset === p ? 'bg-indigo-500 text-white' : 'bg-white/5 text-neutral-400 hover:text-white'
                }`}
              >
                {p}
              </button>
            ))}
          </div>
        </div>

        <div className="flex justify-between items-center pt-2 px-2">
          {eqBands.map((val, idx) => (
            <div key={idx} className="flex flex-col items-center space-y-2">
              <span className="text-[10px] font-mono text-neutral-400">{val > 0 ? `+${val}` : val}дБ</span>
              <input 
                type="range" 
                min="-10" 
                max="10" 
                value={val}
                onChange={(e) => {
                  const newBands = [...eqBands];
                  newBands[idx] = Number(e.target.value);
                  setEqBands(newBands);
                }}
                className="h-28 w-1.5 bg-neutral-800 rounded-lg appearance-none cursor-pointer accent-indigo-500"
                style={{ writingMode: 'bt-lr', appearance: 'slider-vertical', accentColor: accentColor }}
              />
              <span className="text-[9px] font-mono text-neutral-400">{bandLabels[idx]}</span>
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
